cmake_minimum_required(VERSION 3.25)
project(MannyRustPackage LANGUAGES NONE)

get_filename_component(repository "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(MANNY_RUST_RELEASE_DIRECTORY "" CACHE PATH "Cargo x86_64-pc-windows-msvc/release directory")
set(MANNY_DUMPBIN_EXECUTABLE "" CACHE FILEPATH "Microsoft dumpbin executable")
if(NOT IS_ABSOLUTE "${MANNY_RUST_RELEASE_DIRECTORY}")
    message(FATAL_ERROR "MANNY_RUST_RELEASE_DIRECTORY must be an absolute path")
endif()
set(MANNY_DLL_FILE "${MANNY_RUST_RELEASE_DIRECTORY}/manny_uploader.dll")
include("${repository}/cmake/VerifyWindowsImage.cmake")
set(MANNY_NOTICES_FILE "${repository}/THIRD_PARTY_NOTICES.md")
include("${repository}/cmake/VerifyWindowsNotices.cmake")

# Cargo also puts this PDB in deps. Select its documented top-level output explicitly.
set(MANNY_SYMBOL_FILE "${MANNY_RUST_RELEASE_DIRECTORY}/manny_uploader.pdb")
include("${repository}/cmake/VerifyWindowsSymbols.cmake")
include("${repository}/cmake/VerifyWindowsSymbolMatch.cmake")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/input" "${CMAKE_BINARY_DIR}/symbols/Release")
file(COPY_FILE "${MANNY_DLL_FILE}" "${CMAKE_BINARY_DIR}/input/manny_uploader.dll" ONLY_IF_DIFFERENT)
file(COPY_FILE "${MANNY_SYMBOL_FILE}" "${CMAKE_BINARY_DIR}/symbols/Release/manny_uploader.pdb"
    ONLY_IF_DIFFERENT)
set(MANNY_SYMBOL_FILE "${CMAKE_BINARY_DIR}/symbols/Release/manny_uploader.pdb")
include("${repository}/cmake/VerifyWindowsSymbols.cmake")
set(MANNY_DLL_FILE "${CMAKE_BINARY_DIR}/input/manny_uploader.dll")
include("${repository}/cmake/VerifyWindowsSymbolMatch.cmake")
include("${repository}/cmake/VerifyWindowsNotices.cmake")

file(STRINGS "${repository}/rust/addon/Cargo.toml" version_line REGEX "^version = \"[0-9]+\\.[0-9]+\\.[0-9]+\"$")
if(NOT version_line MATCHES "^version = \"([0-9]+\\.[0-9]+\\.[0-9]+)\"$")
    message(FATAL_ERROR "Expected one numeric addon package version")
endif()
set(CPACK_PACKAGE_VERSION "${CMAKE_MATCH_1}")
install(FILES "${CMAKE_BINARY_DIR}/input/manny_uploader.dll" DESTINATION .)
set(CPACK_GENERATOR ZIP)
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY OFF)
set(CPACK_MONOLITHIC_INSTALL ON)
set(CPACK_PACKAGE_CHECKSUM SHA256)
set(CPACK_PACKAGE_DIRECTORY "${CMAKE_BINARY_DIR}/package")
set(CPACK_PACKAGE_FILE_NAME "GW2-Manny-Uploader-${CPACK_PACKAGE_VERSION}-windows-x64")
set(CPACK_PACKAGE_NAME "GW2 Manny Uploader")
set(CPACK_PACKAGE_VENDOR "LoganWal")
include(CPack)
